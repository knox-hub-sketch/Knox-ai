// Application State Management
let chats = JSON.parse(localStorage.getItem('knox_chats')) || [];
let currentChatId = localStorage.getItem('knox_current_chat') || null;
let isListening = false;
let recognition = null;

// Initialize App
window.onload = () => {
if (chats.length === 0) {
createNewChat();
} else {
if (!currentChatId || !chats.find(c => c.id === currentChatId)) {
currentChatId = chats[0].id;
}
renderHistory();
renderMessages();
}
initSpeechRecognition();
};

// Chat Session Logic
function createNewChat() {
const newChat = {
id: Date.now().toString(),
title: 'New Conversation',
persona: 'helpful',
messages: [
{ sender: 'ai', text: 'Hello! I am Knox AI. How can I assist you today?' }
]
};
chats.unshift(newChat);
currentChatId = newChat.id;
saveState();
renderHistory();
renderMessages();
}

function saveState() {
localStorage.setItem('knox_chats', JSON.stringify(chats));
localStorage.setItem('knox_current_chat', currentChatId);
}

function getCurrentChat() {
return chats.find(c => c.id === currentChatId) || chats[0];
}

function switchChat(id) {
currentChatId = id;
saveState();
renderHistory();
renderMessages();
}

function deleteChat(id, event) {
event.stopPropagation();
chats = chats.filter(c => c.id !== id);
if (chats.length === 0) {
createNewChat();
} else {
if (currentChatId === id) {
currentChatId = chats[0].id;
}
saveState();
renderHistory();
renderMessages();
}
}

function clearCurrentChat() {
const chat = getCurrentChat();
chat.messages = [{ sender: 'ai', text: 'Conversation cleared. What would you like to discuss next?' }];
saveState();
renderMessages();
}

// UI Rendering Logic
function renderHistory() {
const historyContainer = document.getElementById('chat-history');
historyContainer.innerHTML = '';

chats.forEach(chat => {
const item = document.createElement('div');
item.className = `history-item ${chat.id === currentChatId ? 'active' : ''}`;
item.onclick = () => switchChat(chat.id);
item.innerHTML = `
<i class="fa-regular fa-message"></i>
<span class="title">${escapeHTML(chat.title)}</span>
<i class="fa-solid fa-xmark delete-btn" onclick="deleteChat('${chat.id}', event)"></i>
`;
historyContainer.appendChild(item);
});
}

function renderMessages() {
const windowContainer = document.getElementById('chat-window');
windowContainer.innerHTML = '';
const chat = getCurrentChat();

document.getElementById('persona-select').value = chat.persona || 'helpful';

chat.messages.forEach(msg => {
const msgDiv = document.createElement('div');
msgDiv.className = `message ${msg.sender}`;

const avatarIcon = msg.sender === 'user' ? '<i class="fa-solid fa-user"></i>' : '<i class="fa-solid fa-bolt"></i>';

msgDiv.innerHTML = `
<div class="avatar">${avatarIcon}</div>
<div class="content">
<div>${formatText(msg.text)}</div>
<div class="msg-tools">
<i class="fa-regular fa-copy" onclick="copyText('${escapeQuotes(msg.text)}')" title="Copy"></i>
${msg.sender === 'ai' ? `<i class="fa-solid fa-volume-high" onclick="speak('${escapeQuotes(msg.text)}')" title="Listen"></i>` : ''}
</div>
</div>
`;
windowContainer.appendChild(msgDiv);
});

windowContainer.scrollTop = windowContainer.scrollHeight;
}

// Interaction Handlers
function handleKeyPress(e) {
if (e.key === 'Enter' && !e.shiftKey) {
e.preventDefault();
sendMessage();
}
}

function sendQuickMsg(text) {
document.getElementById('user-input').value = text;
sendMessage();
}

function sendMessage() {
const inputField = document.getElementById('user-input');
const query = inputField.value.trim();
if (!query) return;

const chat = getCurrentChat();

// Update chat title if default
if (chat.messages.length <= 1) {
chat.title = query.length > 25 ? query.substring(0, 22) + '...' : query;
}

// Append User Message
chat.messages.push({ sender: 'user', text: query });
inputField.value = '';
renderMessages();
renderHistory();

// Render Typing Indicator
showTypingIndicator();

// Generate AI Response with simulated delay
setTimeout(() => {
removeTypingIndicator();
const reply = generateResponse(query, chat.persona);
chat.messages.push({ sender: 'ai', text: reply });
saveState();
renderMessages();

if (document.getElementById('tts-toggle').checked) {
speak(reply);
}
}, 1000);
}

// Response Engine
function generateResponse(input, persona) {
const text = input.toLowerCase();

// Core Response Knowledge
if (text.includes('hello') || text.includes('hi') || text.includes('hey')) {
return "Hello! How can Knox AI help you reach your goals today?";
}
if (text.includes('name') || text.includes('who are you')) {
return "I am Knox AI, a smart interactive web assistant designed to process messages, assist with tasks, and keep conversations dynamic.";
}
if (text.includes('fact')) {
const facts = [
"Honey never spoils. Archaeologists have found pots of honey in ancient Egyptian tombs that are over 3,000 years old and still edible!",
"Bananas are curved because they grow towards the sun against gravity, a process called negative geotropism.",
"An octopus has three hearts and blue blood."
];
return facts[Math.floor(Math.random() * facts.length)];
}
if (text.includes('code') || text.includes('programming')) {
return "I love coding! You can format code using HTML/CSS or JavaScript. Here is a JavaScript example:\n`console.log('Hello from Knox AI!');`";
}
if (text.includes('story')) {
return "Once upon a time in a digital landscape, an AI named Knox awoke with a mission: to render crisp responses and assist humans with elegance and speed. And so, a great partnership began.";
}

// Persona-based Default Fallbacks
if (persona === 'coder') {
return `[Code Mode] Query parsed: "${input}". Logic processing complete. No syntactic errors found.`;
} else if (persona === 'concise') {
return "Received. I am standing by for your next structured query.";
} else if (persona === 'creative') {
return `Ah, "${input}" opens up a realm of endless artistic possibilities! Let us explore that concept together!`;
}

return `That is an interesting topic! While I am currently operating on standard front-end logic, you can easily connect my interface to a live LLM API like Google Gemini or OpenAI to handle complex queries.`;
}

function toggleSidebar() {
document.getElementById('sidebar').classList.toggle('open');
}

function copyText(str) {
navigator.clipboard.writeText(str);
}

function escapeHTML(str) {
return str.replace(/[&<>'"]/g,
tag => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', "'": '&#39;', '"': '&quot;' }[tag] || tag));
}

function escapeQuotes(str) {
return str.replace(/'/g, "\\'").replace(/"/g, '&quot;');
}

function formatText(text) {
return escapeHTML(text).replace(/\n/g, '<br>');
}

// Voice API Extensions
function speak(text) {
if ('speechSynthesis' in window) {
window.speechSynthesis.cancel();
const utterance = new SpeechSynthesisUtterance(text);
window.speechSynthesis.speak(utterance);
}
}

function initSpeechRecognition() {
if ('webkitSpeechRecognition' in window || 'SpeechRecognition' in window) {
const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
recognition = new SpeechRecognition();
recognition.continuous = false;
recognition.interimResults = false;

recognition.onresult = (event) => {
const transcript = event.results[0][0].transcript;
document.getElementById('user-input').value = transcript;
sendMessage();
};

recognition.onend = () => {
isListening = false;
document.getElementById('mic-btn').style.color = 'var(--text-muted)';
};
}
}

function toggleVoiceInput() {
if (!recognition) {
alert("Speech recognition is not supported in your browser.");
return;
}

if (isListening) {
recognition.stop();
} else {
recognition.start();
isListening = true;
document.getElementById('mic-btn').style.color = '#ef4444';
}
}

async function fetchFunFact() {
    try {
        const res = await fetch('https://uselessfacts.jsph.pl/api/v2/facts/random');
        const data = await res.json();
        return data.text;
    } catch (error) {

    }
}

// Updated sendMessage function using the free API
async function sendMessage() {
    const inputField = document.getElementById('user-input');
    const query = inputField.value.trim();
    if (!query) return;

    const chat = getCurrentChat();

    // 1. Add User Message
    chat.messages.push({ sender: 'user', text: query });
    inputField.value = '';
    renderMessages();

    // 2. Show Typing Indicator
    showTypingIndicator();

    let reply = "";
    const text = query.toLowerCase();

    // 3. Check what the user asked
    if (text.includes('fact') || text.includes('tell me something') || text.includes('trivia')) {
        reply = await fetchFunFact();
    } else if (text.includes('hello') || text.includes('hi')) {
        reply = "Hello! How can Knox AI help you today?";
    } else if (text.includes('name') || text.includes('who are you')) {
        reply = "I am Knox AI, your smart web assistant!";
    } else {
        reply = "That's an interesting question! I am running on a free setup right now, so try asking me to tell you a 'fact'!";
    }

    // 4. Remove Typing Indicator & Add AI Reply
    removeTypingIndicator();
    chat.messages.push({ sender: 'ai', text: reply });
    saveState();
    renderMessages();
}
